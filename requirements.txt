# Lingua — standalone language-learning app.
# Intentionally separate from the root/apps/core requirements: this app ships
# and deploys independently of ARIA.
#
# Runtime/production dependencies only — this is exactly what ships inside
# the Docker image (see Dockerfile). Test-only tooling lives in
# requirements-dev.txt instead, so `pip install -r requirements.txt` never
# pulls a test framework into production.
#
# python-multipart was removed from this file: FastAPI only needs it to
# parse multipart/form-data (Form()/UploadFile params), and nothing in this
# app uses either — audio reaches the backend as a raw request body
# (routers/content.py's speech_to_text) or base64 over the WebSocket
# (routers/conversation.py), never as a multipart upload.
fastapi==0.141.1
uvicorn[standard]==0.37.0
pydantic==2.13.4
httpx==0.28.1

# Durable production storage (Supabase Postgres) — see backend/db.py
psycopg[binary]==3.3.4

# Self-hosted, free, unlimited TTS — see backend/piper_tts.py for why 1.6.0
# specifically (GPL-3.0-or-later; the last MIT release has no Linux/py3.12
# wheel for its phonemizer dependency, so it doesn't install on this image).
piper-tts==1.6.0

# Optional per-persona voice tier (backend/personas.py, hf_client.py) — only
# exercised when a persona's voice_description is set; Piper above stays
# the universal default. gradio_client talks to the verified Parler-TTS
# Space fallback; num2words preprocesses digits for that model's input.
gradio_client==2.6.0
num2words==0.5.14
